    <script>
        const myName = localStorage.getItem('shepot_my_name') || 'Я';
        const partnerName = localStorage.getItem('shepot_partner_name') || 'Партнёр';

        let myFantasies = JSON.parse(localStorage.getItem('shepot_my_fantasies') || '[]');
        let partnerFantasies = JSON.parse(localStorage.getItem('shepot_partner_fantasies') || '[]');
        let currentTab = 'my'; // 'my' или 'partner'

        function switchTab(tab) {
            currentTab = tab;
            
            document.getElementById('switchMy').classList.toggle('active', tab === 'my');
            document.getElementById('switchPartner').classList.toggle('active', tab === 'partner');
            
            document.getElementById('descMy').style.display = tab === 'my' ? 'block' : 'none';
            document.getElementById('descPartner').style.display = tab === 'partner' ? 'block' : 'none';
            
            // Поле ввода показываем только в "мои фантазии"
            document.getElementById('inputRow').style.display = tab === 'my' ? 'flex' : 'none';
            
            renderFantasies();
        }

        function sendFantasy() {
            const input = document.getElementById('fantasyInput');
            const text = input.value.trim();
            
            if (!text) {
                showToast('⚠️ Напишите фантазию');
                return;
            }
            
            if (text.length < 5) {
                showToast('⚠️ Слишком коротко. Опишите подробнее');
                return;
            }
            
            // Добавляем фантазию
            const fantasy = {
                id: Date.now(),
                text: text,
                date: new Date().toISOString(),
                from: 'me',
                response: null // 'agreed', 'discuss', 'declined'
            };
            
            myFantasies.push(fantasy);
            localStorage.setItem('shepot_my_fantasies', JSON.stringify(myFantasies));
            
            // Отправляем партнёру
            partnerFantasies.push({
                ...fantasy,
                from: 'partner'
            });
            localStorage.setItem('shepot_partner_fantasies', JSON.stringify(partnerFantasies));
            
            input.value = '';
            renderFantasies();
            showToast('✨ Фантазия отправлена!');
        }

        function respondToFantasy(id, response) {
            // Обновляем в фантазиях партнёра
            const partnerIndex = partnerFantasies.findIndex(f => f.id === id);
            if (partnerIndex !== -1) {
                partnerFantasies[partnerIndex].response = response;
                localStorage.setItem('shepot_partner_fantasies', JSON.stringify(partnerFantasies));
            }
            
            // Обновляем в фантазиях автора
            const myIndex = myFantasies.findIndex(f => f.id === id);
            if (myIndex !== -1) {
                myFantasies[myIndex].response = response;
                localStorage.setItem('shepot_my_fantasies', JSON.stringify(myFantasies));
            }
            
            renderFantasies();
            showToast('✅ Ответ отправлен!');
        }

        function deleteFantasy(id) {
            if (!confirm('Удалить эту фантазию?')) return;
            
            myFantasies = myFantasies.filter(f => f.id !== id);
            partnerFantasies = partnerFantasies.filter(f => f.id !== id);
            
            localStorage.setItem('shepot_my_fantasies', JSON.stringify(myFantasies));
            localStorage.setItem('shepot_partner_fantasies', JSON.stringify(partnerFantasies));
            
            renderFantasies();
            showToast('🗑️ Фантазия удалена');
        }

        function getResponseInfo(response) {
            const map = {
                'agreed': { text: '✅ Согласен(а)', class: 'response-agreed', emoji: '🎉' },
                'discuss': { text: '💬 Надо обсудить', class: 'response-discuss', emoji: '🤔' },
                'declined': { text: '❌ Не согласен(а)', class: 'response-declined', emoji: '😔' }
            };
            return map[response] || null;
        }

        function renderFantasies() {
            const list = document.getElementById('fantasiesList');
            
            if (currentTab === 'my') {
                // Мои фантазии
                if (myFantasies.length === 0) {
                    list.innerHTML = `
                        <div class="empty-state">
                            <div class="empty-icon">💭</div>
                            <div>У вас пока нет фантазий</div>
                            <div style="font-size:10px;opacity:0.7;margin-top:4px;">Напишите первую — партнёр увидит и ответит</div>
                        </div>
                    `;
                    return;
                }
                
                list.innerHTML = myFantasies.slice().reverse().map(f => {
                    const responseInfo = getResponseInfo(f.response);
                    const date = new Date(f.date).toLocaleDateString('ru-RU');
                    
                    return `
                        <div class="fantasy-card ${f.response ? 'responded' : ''}">
                            <button class="fantasy-delete" onclick="deleteFantasy(${f.id})" title="Удалить">×</button>
                            <div class="fantasy-label">Моя фантазия</div>
                            <div class="fantasy-text">${escapeHtml(f.text)}</div>
                            <div class="fantasy-meta">📅 ${date}</div>
                            ${f.response ? `
                                <div class="fantasy-response">
                                    <div class="response-label ${responseInfo.class}">
                                        Ответ партнёра: ${responseInfo.emoji} ${responseInfo.text}
                                    </div>
                                </div>
                            ` : `
                                <div class="fantasy-response">
                                    <div class="response-label" style="color:#FF9800;">
                                        ⏳ Ожидает ответа партнёра...
                                    </div>
                                </div>
                            `}
                        </div>
                    `;
                }).join('');
            } else {
                // Фантазии партнёра
                const partnerOnly = partnerFantasies.filter(f => f.from === 'partner');
                
                if (partnerOnly.length === 0) {
                    list.innerHTML = `
                        <div class="empty-state">
                            <div class="empty-icon">💭</div>
                            <div>Партнёр пока не поделился фантазиями</div>
                            <div style="font-size:10px;opacity:0.7;margin-top:4px;">Как только партнёр напишет — вы увидите здесь</div>
                        </div>
                    `;
                    return;
                }
                
                list.innerHTML = partnerOnly.slice().reverse().map(f => {
                    const responseInfo = getResponseInfo(f.response);
                    const date = new Date(f.date).toLocaleDateString('ru-RU');
                    
                    return `
                        <div class="fantasy-card ${f.response ? 'responded' : ''}">
                            <div class="fantasy-label">Фантазия ${partnerName}</div>
                            <div class="fantasy-text">${escapeHtml(f.text)}</div>
                            <div class="fantasy-meta">📅 ${date}</div>
                            ${f.response ? `
                                <div class="fantasy-response">
                                    <div class="response-label ${responseInfo.class}">
                                        Ваш ответ: ${responseInfo.emoji} ${responseInfo.text}
                                    </div>
                                </div>
                            ` : `
                                <div class="fantasy-response">
                                    <div class="response-label" style="color:var(--gold);margin-bottom:8px;">
                                        Ваш ответ:
                                    </div>
                                    <div class="response-buttons">
                                        <button class="btn-response agree" onclick="respondToFantasy(${f.id}, 'agreed')">✅ Согласен(а)</button>
                                        <button class="btn-response discuss" onclick="respondToFantasy(${f.id}, 'discuss')">💬 Обсудить</button>
                                        <button class="btn-response decline" onclick="respondToFantasy(${f.id}, 'declined')">❌ Не согласен(а)</button>
                                    </div>
                                </div>
                            `}
                        </div>
                    `;
                }).join('');
            }
        }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        function showToast(message) {
            const existing = document.querySelector('.toast');
            if (existing) existing.remove();
            
            const toast = document.createElement('div');
            toast.className = 'toast';
            toast.textContent = message;
            document.body.appendChild(toast);
            
            setTimeout(() => toast.remove(), 3000);
        }

        // Инициализация
        switchTab('my');
    </script>
