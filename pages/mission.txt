    <script>
        const myName = localStorage.getItem('shepot_my_name') || 'Я';
        const partnerName = localStorage.getItem('shepot_partner_name') || 'Партнёр';

        // Хранилище: { myWishes: [{id, text, done}], partnerWishes: [{id, text, done}] }
        let data = JSON.parse(localStorage.getItem('shepot_mission_data')) || {
            myWishes: [],
            partnerWishes: []
        };

        let wishIdCounter = 0;

        function getNextId() {
            const allWishes = [...data.myWishes, ...data.partnerWishes];
            return allWishes.reduce((max, w) => Math.max(max, w.id || 0), 0) + 1;
        }

        function saveData() {
            localStorage.setItem('shepot_mission_data', JSON.stringify(data));
        }

        function render() {
            renderMyWishes();
            renderPartnerWishes();
            updateProgress();
            checkCelebration();
        }

        function renderMyWishes() {
            const container = document.getElementById('myWishes');
            const list = data.myWishes;

            if (list.length === 0) {
                container.innerHTML = '<div class="empty-text">Пока нет желаний</div>';
            } else {
                container.innerHTML = list.map((wish, index) => `
                    <div class="wish-item ${wish.done ? 'done' : ''}">
                        <div class="wish-check" onclick="toggleMyWish(${index})">${wish.done ? '✓' : ''}</div>
                        <div class="wish-text">${escapeHtml(wish.text)}</div>
                        <button class="wish-delete" onclick="deleteMyWish(${index})">×</button>
                    </div>
                `).join('');
            }

            const total = list.length;
            const done = list.filter(w => w.done).length;
            document.getElementById('myCount').textContent = `${done} из ${total}`;
            document.getElementById('myRemaining').textContent = Math.max(0, 7 - total);
            document.getElementById('addMyWishBtn').disabled = total >= 7;
        }

        function renderPartnerWishes() {
            const container = document.getElementById('partnerWishes');
            const list = data.partnerWishes;

            if (list.length === 0) {
                container.innerHTML = '<div class="empty-text">Партнёр ещё не написал желания</div>';
            } else {
                container.innerHTML = list.map((wish, index) => `
                    <div class="wish-item ${wish.done ? 'done' : ''}">
                        <div class="wish-check" onclick="togglePartnerWish(${index})">${wish.done ? '✓' : ''}</div>
                        <div class="wish-text">${escapeHtml(wish.text)}</div>
                        <div class="wish-author">для меня</div>
                    </div>
                `).join('');
            }

            const total = list.length;
            const done = list.filter(w => w.done).length;
            document.getElementById('partnerCount').textContent = `${done} из ${total}`;
        }

        function updateProgress() {
            const allWishes = data.partnerWishes;
            const total = allWishes.length;
            const done = allWishes.filter(w => w.done).length;

            document.getElementById('progressText').textContent = `${done} из ${total}`;
            const percent = total > 0 ? (done / total) * 100 : 0;
            document.getElementById('progressFill').style.width = `${Math.min(percent, 100)}%`;

            if (total === 0) {
                document.getElementById('progressStatus').textContent = 'Напишите желания для партнёра';
            } else if (done === total && total > 0) {
                document.getElementById('progressStatus').textContent = '🎉 Все желания выполнены!';
            } else {
                document.getElementById('progressStatus').textContent = `Выполнено ${done} из ${total} желаний`;
            }
        }

        function checkCelebration() {
            const total = data.partnerWishes.length;
            const done = data.partnerWishes.filter(w => w.done).length;
            const celebration = document.getElementById('celebration');
            const mainScreen = document.getElementById('mainScreen');

            if (total >= 7 && done === total) {
                celebration.classList.add('show');
                mainScreen.style.display = 'none';
            } else {
                celebration.classList.remove('show');
                mainScreen.style.display = 'flex';
            }
        }

        // === Мои желания ===

        function addMyWish() {
            const input = document.getElementById('myWishInput');
            const text = input.value.trim();

            if (!text) {
                showModal('Пустое желание', 'Напиши, что ты хочешь от партнёра.');
                return;
            }

            if (data.myWishes.length >= 7) {
                showModal('Лимит достигнут', 'Ты уже написал(а) 7 желаний. Дождись, пока партнёр выполнит их.');
                return;
            }

            data.myWishes.push({
                id: getNextId(),
                text: text,
                done: false
            });

            saveData();
            input.value = '';
            render();
        }

        function toggleMyWish(index) {
            const wish = data.myWishes[index];
            // Нельзя отметить как выполненное своё же желание — это делает партнёр
            showModal(
                'Отметить выполненным?',
                `Ты хочешь отметить желание "${wish.text}" как выполненное? Это действие выполняет партнёр, когда делает что-то для тебя.`,
                function() {
                    // Это логика для партнёра, но в демо-режиме разрешаем
                    wish.done = !wish.done;
                    saveData();
                    render();
                }
            );
        }

        function deleteMyWish(index) {
            const wish = data.myWishes[index];
            if (wish.done) {
                showModal('Нельзя удалить', 'Выполненное желание нельзя удалить.');
                return;
            }

            showModal(
                'Удалить желание?',
                `Ты уверен(а), что хочешь удалить желание "${wish.text}"?`,
                function() {
                    data.myWishes.splice(index, 1);
                    saveData();
                    render();
                }
            );
        }

        // === Желания партнёра ===

        function togglePartnerWish(index) {
            const wish = data.partnerWishes[index];
            wish.done = !wish.done;
            saveData();
            render();
        }

        // === Сброс ===

        function resetWishes() {
            showModal(
                'Новый круг желаний',
                'Все желания будут удалены. Ты сможешь написать новые 7 желаний для партнёра.',
                function() {
                    data.myWishes = [];
                    data.partnerWishes = [];
                    saveData();
                    render();
                    document.getElementById('celebration').classList.remove('show');
                    document.getElementById('mainScreen').style.display = 'flex';
                }
            );
        }

        // === Модалка ===

        let modalCallback = null;

        function showModal(title, desc, callback) {
            document.getElementById('modalTitle').textContent = title;
            document.getElementById('modalDesc').textContent = desc;
            document.getElementById('modal').classList.add('show');

            if (callback) {
                modalCallback = callback;
                document.getElementById('modalConfirmBtn').style.display = 'block';
                document.getElementById('modalConfirmBtn').onclick = function() {
                    closeModal();
                    if (modalCallback) {
                        modalCallback();
                        modalCallback = null;
                    }
                };
            } else {
                document.getElementById('modalConfirmBtn').style.display = 'none';
            }
        }

        function closeModal() {
            document.getElementById('modal').classList.remove('show');
            modalCallback = null;
        }

        // === Утилиты ===

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        function navigate(page) {
            const pages = {
                'today': 'today.html',
                'questions': 'questions.html',
                'games': 'games.html',
                'beyond': 'beyond.html',
                'moments': 'moments.html',
                'we': 'we.html',
                'main': 'main.html'
            };
            if (pages[page]) {
                window.location.href = pages[page];
            }
        }

        // Инициализация
        render();
    </script>
