        // ОТРИСОВКА КОЛЕСА

        const canvas = document.getElementById('wheelCanvas');
        const ctx = canvas.getContext('2d');
        const centerX = 200;
        const centerY = 200;
        const radius = 180;

        let currentRotation = 0;
        let isSpinning = false;

        function drawWheel(rotation) {
            const sectorAngle = (2 * Math.PI) / SECTORS.length;
            ctx.clearRect(0, 0, 400, 400);

            SECTORS.forEach((sector, index) => {
                const startAngle = index * sectorAngle + rotation;
                const endAngle = startAngle + sectorAngle;

                // Рисуем сектор
                ctx.beginPath();
                ctx.moveTo(centerX, centerY);
                ctx.arc(centerX, centerY, radius, startAngle, endAngle);
                ctx.closePath();
                ctx.fillStyle = sector.color;
                ctx.fill();
                ctx.strokeStyle = '#0D1117';
                ctx.lineWidth = 3;
                ctx.stroke();

                // Текст в секторе
                const textAngle = startAngle + sectorAngle / 2;
                const textRadius = radius * 0.65;
                const x = centerX + Math.cos(textAngle) * textRadius;
                const y = centerY + Math.sin(textAngle) * textRadius;

                ctx.save();
                ctx.translate(x, y);
                ctx.rotate(textAngle + Math.PI / 2);
                ctx.font = 'bold 18px Inter, sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillStyle = '#FFFFFF';
                ctx.shadowColor = 'rgba(0,0,0,0.5)';
                ctx.shadowBlur = 8;
                ctx.fillText(sector.emoji, 0, 0);
                ctx.restore();

                // Название категории
                const labelRadius = radius * 0.85;
                const lx = centerX + Math.cos(textAngle) * labelRadius;
                const ly = centerY + Math.sin(textAngle) * labelRadius;
                ctx.save();
                ctx.translate(lx, ly);
                ctx.rotate(textAngle + Math.PI / 2);
                ctx.font = 'bold 12px Inter, sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillStyle = '#FFFFFF';
                ctx.shadowColor = 'rgba(0,0,0,0.5)';
                ctx.shadowBlur = 6;
                ctx.fillText(sector.label, 0, 0);
                ctx.restore();
            });

            // Внешняя обводка
            ctx.beginPath();
            ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);
            ctx.strokeStyle = 'rgba(255,255,255,0.1)';
            ctx.lineWidth = 4;
            ctx.stroke();

            // Декоративные точки
            for (let i = 0; i < SECTORS.length; i++) {
                const angle = i * sectorAngle + sectorAngle / 2 + rotation;
                const dotR = radius - 12;
                const dx = centerX + Math.cos(angle) * dotR;
                const dy = centerY + Math.sin(angle) * dotR;
                ctx.beginPath();
                ctx.arc(dx, dy, 4, 0, 2 * Math.PI);
                ctx.fillStyle = 'rgba(255,255,255,0.3)';
                ctx.fill();
            }
        }

        drawWheel(0);

        // ВРАЩЕНИЕ КОЛЕСА
        // ============================================================

        function spinWheel() {
            if (isSpinning) return;
            isSpinning = true;
            document.getElementById('spinBtn').disabled = true;

            // Случайный сектор
            const targetIndex = Math.floor(Math.random() * SECTORS.length);

            // Случайное количество оборотов (5-8)
            const extraSpins = 5 + Math.random() * 3;

            // Угол одного сектора
            const sectorAngle = (2 * Math.PI) / SECTORS.length;

            // Целевой угол: чтобы сектор оказался под стрелкой (сверху = -90 градусов)
            // Стрелка указывает вверх, то есть на угол -PI/2
            const pointerAngle = -Math.PI / 2;

            // Вычисляем угол для целевого сектора
            // Нужно, чтобы центр сектора был под стрелкой
            const targetAngle = targetIndex * sectorAngle + sectorAngle / 2;

            // Поворот: чтобы targetAngle оказался в позиции pointerAngle
            // currentRotation + targetAngle = pointerAngle + 2*PI*n
            // currentRotation = pointerAngle - targetAngle + 2*PI*n
            let newRotation = pointerAngle - targetAngle;

            // Добавляем полные обороты
            newRotation += extraSpins * 2 * Math.PI;

            // Нормализуем, чтобы вращение было в плюс
            while (newRotation < currentRotation) {
                newRotation += 2 * Math.PI;
            }

            // Плавная анимация
            const canvasEl = document.getElementById('wheelCanvas');
            const startRotation = currentRotation;
            const duration = 4000 + Math.random() * 1000;
            const startTime = performance.now();

            function animate(timestamp) {
                const elapsed = timestamp - startTime;
                const progress = Math.min(elapsed / duration, 1);

                // Easing для плавного замедления
                const ease = 1 - Math.pow(1 - progress, 3);
                const current = startRotation + (newRotation - startRotation) * ease;

                drawWheel(current);

                if (progress < 1) {
                    requestAnimationFrame(animate);
                } else {
                    currentRotation = newRotation;
                    isSpinning = false;
                    document.getElementById('spinBtn').disabled = false;

                    // Показываем результат
                    showResult(targetIndex);
                }
            }

            requestAnimationFrame(animate);
        }

        // ПОКАЗ РЕЗУЛЬТАТА
        // ============================================================

        function showResult(sectorIndex) {
            const sector = SECTORS[sectorIndex];
            const ideas = IDEAS[sector.id] || [];
            const idea = ideas[Math.floor(Math.random() * ideas.length)];

            if (!idea) {
                backToWheel();
                return;
            }

            document.getElementById('mainContent').style.display = 'none';
            document.getElementById('resultScreen').classList.add('active');

            document.getElementById('resultIcon').textContent = sector.emoji;
            document.getElementById('resultCategoryLabel').textContent = sector.emoji + ' ' + sector.label;
            document.getElementById('resultCategory').textContent = sector.label.toUpperCase();
            document.getElementById('resultTitle').textContent = idea.title;

            document.getElementById('resultTags').innerHTML = `
                <span class="tag">⏱️ ${idea.duration}</span>
                <span class="tag">${idea.place}</span>
            `;

            document.getElementById('resultTip').innerHTML = '<strong>Что нужно:</strong> ' + idea.tip;
        }

        function backToWheel() {
            document.getElementById('mainContent').style.display = 'block';
            document.getElementById('resultScreen').classList.remove('active');
        }
